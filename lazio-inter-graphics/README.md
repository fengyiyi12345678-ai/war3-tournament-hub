## 🚀 导出 PNG 一键命令

```bash
# 用 rsvg-convert（macOS: brew install librsvg）
for f in *.svg; do
  rsvg-convert -w 1080 -h 1350 "$f" -o "${f%.svg}.png"
done

# 或 Inkscape
for f in *.svg; do
  inkscape "$f" --export-type=png --export-filename="${f%.svg}.png" -w 1080 -h 1350
done

# 或 ImageMagick（需要 librsvg 后端）
for f in *.svg; do
  magick -density 144 "$f" -resize 1080x1350 "${f%.svg}.png"
done
```
